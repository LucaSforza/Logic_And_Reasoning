\section{Introduction: From Kripke Models to Game Frames}

\begin{frame}{Single-Agent Modal Logic}
	\begin{itemize}
		\item In the simplest case, we have \textbf{one agent} who can choose between actions. (Normal Model Logic models) % TODO: migliorare spiegazione
		\item A \textbf{Kripke model} $(S, R)$ uses an accessibility relation $R$ that associates to every state all reachable states.
		\item A modal formula $\Diamond \varphi$ means: ``the agent can act so that $\varphi$ will be true.''
	\end{itemize}

	\vspace{0.5em}
	Formally: $M, s \models \Diamond \varphi \iff \exists t \in S: sRt \text{ and } M, t \models \varphi$.
\end{frame}

\begin{frame}{The Multi-Agent Problem}
	\begin{itemize}
		\item Extending to multiple agents with individual accessibility relations $R_i$ considers agents \textbf{in isolation}.
		\item But agents' actions \textbf{interact}: what happens when agent 1 and agent 2 act \textbf{simultaneously}?
		\item The outcome depends on the \textbf{combination} of all participants' choices.
	\end{itemize}

	\vspace{0.5em}
	Standard multi-agent modal logic \textbf{cannot} capture this interaction.
\end{frame}

\begin{frame}{The Solution: Game Frames (Pauly 2002)}
	\begin{block}{Definition (Game Frame)}
		Marc Pauly introduces \textbf{game frames}: at every state, a \textbf{strategic game} is played among the agents.
		Outcomes of the game are states of the model.
	\end{block}

	\begin{itemize}
		\item Game frames are essentially \textbf{extensive games with simultaneous moves}.
		\item They generalize both Kripke models (1-player case) and non-normal modal logics (2-player case).
	\end{itemize}
\end{frame}

\begin{frame}{Strategic Games}
	A strategic game $G = (N, \{\Sigma_i \mid i \in N\}, o, S)$ consists of:
	\begin{itemize}
		\item $N$: finite set of agents (players)
		\item $\Sigma_i$: set of strategies for agent $i$
		\item $S$: set of outcome states
		\item $o: \prod_{i \in N} \Sigma_i \rightarrow S$: outcome function
	\end{itemize}

	\vspace{0.5em}
	We consider \emph{game forms} only: no preference relations are assumed.
\end{frame}

\begin{frame}{Game Frames}
	\begin{block}{Definition}
		Let $\Gamma^N_S$ be the set of all strategic games over agents $N$ and states $S$.
		A \emph{game frame} is a pair $(S, \gamma)$ where $\gamma: S \rightarrow \Gamma^N_S$ assigns a strategic game to each state.
	\end{block}

	\begin{itemize}
		\item At every state, agents play a game; the outcome leads to a new state (and a new game).
		\item This is a \textbf{total function}: every state has an associated game.
	\end{itemize}
\end{frame}

\begin{frame}{From Strategy to Power: Effectivity Functions}
	\begin{block}{Definition}
		The \emph{effectivity function} $E_G: \mathcal{P}(N) \rightarrow \mathcal{P}(\mathcal{P}(S))$ derived from game $G$:
		\[ X \in E_G(C) \iff \exists \sigma_C \forall \sigma_{N \setminus C}: o(\sigma_C, \sigma_{N \setminus C}) \in X \]
	\end{block}

	\begin{itemize}
		\item Coalition $C$ is \textbf{effective} for $X$ if $C$ has a joint strategy that \textbf{forces} the outcome into $X$, regardless of what the others do.
		\item The notion of effectiveness can also be investigated more generally. For a set of agents $N$ and a set of outcomes $S$, we call any
		      function $E: \mathcal{P}(N) \rightarrow \mathcal{P}(\mathcal{P}(S))$ an effectivity function.
	\end{itemize}
\end{frame}

\begin{frame}{Playability (Pauly 2002)}
	It is natural to require them to
	satisfy certain properties and which properties we require will depend on the exact situation
	we want to model. % TODO: grammatica

	\begin{description}
		\item[Outcome-monotonicity] $X \in E(C, s)$ and $X \subseteq X' \Rightarrow X' \in E(C, s)$.
		\item[Coalition-monotonicity] for $C \subseteq C' \subseteq N$, $E(C) \subseteq E(C')$
		\item[$C$-Regular] $\forall X$ if $X \in E(C) \implies \overline{X} \notin E(\overline{C})$
		\item[$C$-Maximal] $\forall X$ if $\overline{X} \notin E(\overline{C}) \implies X \in E(C)$
		\item[Regular (Maximal)] if $\forall C \subseteq N$ $E$ is $C$-Regular (Maximal)
		\item[Superadditivity] $C_1 \cap C_2 = \emptyset$: $X_1 \in E(C_1)$, $X_2 \in E(C_2) \Rightarrow X_1 \cap X_2 \in E(C_1 \cup C_2)$.
	\end{description}

\end{frame}

\begin{frame}{Playability (Pauly 2002)} % TODO: grammatica
	The choice of these properties is based on the situation we want to model.
	In our case the question to be examined is which effectivity functions are effectivity functions of
	some strategic game.

	An effectivity function is \textbf{playable} if:

	\begin{description}
		\item $\forall C \subseteq N$ $\emptyset$ \notin $E(C)$
		\item $\forall C \subseteq N$ $S \in E(C)$
		\item $E$ is $N$-maximal
		\item $E$ is outcome-monotonic
		\item $E$ is superadditive
	\end{description}

\end{frame}

		$E$ is playable $\iff$ $\exists G$ such that $G$ is a strategic game and $E_G = E$

\begin{frame}{From Coalition Frames to Models}
	Let $E$ be $E: S \rightarrow \mathcal{P}(N) -> \mathcal{P}(\mathcal{P}(S))$. % associate every state to a playable effectivity function

	\begin{description}
		\item[Coalition Frame] A pair $\mathcal{F} = (S, E)$ where $E$ assigns a playable effectivity function to each state.
		\item[Coalition Model] A triple $\mathcal{M} = (S, E, \pi)$ adding a valuation $\pi$.
	\end{description}

	\vspace{0.5em}
	The semantics:
	\[ \mathcal{M}, s \models \langle\!\langle C \rangle\!\rangle \varphi \iff \varphi^{\mathcal{M}} \in E(s, C) \]
	where $\varphi^{\mathcal{M}} = \{t \in S \mid \mathcal{M}, t \models \varphi\}$.
\end{frame}

\note{
	characterization theorem assure that Coalition Frames are just Game frames.
}

\begin{frame}{CL Axiomatization (Ågotnes et al. 2008, Table 1)}
	Sound and complete for weak playability models:

	\begin{description}
		\item[$\mathbf{Ag\bot}$] $\neg \langle\!\langle \mathcal{A}g \rangle\!\rangle \bot$
		\item[$\mathbf{\top}$] $\neg \langle\!\langle \emptyset \rangle\!\rangle \bot \to \langle\!\langle C \rangle\!\rangle \top$
		\item[$\mathbf{\bot}$] $\langle\!\langle C \rangle\!\rangle \bot \to \langle\!\langle C' \rangle\!\rangle \bot$ \quad ($C' \subseteq C$)
		\item[$\mathbf{Ag}$] $\neg \langle\!\langle \emptyset \rangle\!\rangle \neg \varphi \to \langle\!\langle \mathcal{A}g \rangle\!\rangle \varphi$
		\item[$\mathbf{S}$] $(\langle\!\langle C_1 \rangle\!\rangle \varphi_1 \land \langle\!\langle C_2 \rangle\!\rangle \varphi_2) \to \langle\!\langle C_1 \cup C_2 \rangle\!\rangle (\varphi_1 \land \varphi_2)$ \quad ($C_1 \cap C_2 = \emptyset$)
	\end{description}

	\vspace{0.3em}
	\rule{\textwidth}{0.4pt}
	\vspace{0.3em}

	\textbf{Rules:} Modus Ponens (MP) and Distribution:
	\[ \vdash \varphi \to \psi \;\Rightarrow\; \vdash \langle\!\langle C \rangle\!\rangle \varphi \to \langle\!\langle C \rangle\!\rangle \psi \]
	plus all propositional tautologies (Prop).
\end{frame}

\note{
	The axiomatization of CL is sound and complete for weak playability models.
	Each axiom encodes one of the semantic playability properties at the logical level.
	The rules are standard: Modus Ponens and Distribution, which is the usual modal
	axiom ensuring the modality distributes over implication.
	The most important axiom for complexity is superadditivity: it is precisely what makes
	satisfiability PSPACE-complete rather than NP-complete.
	(Without it, each $\langle\!\langle C \rangle\!\rangle \varphi$ requires just one recursive check; with it, any subset of modal
	formulas can be combined, forcing $2^n$ recursive calls.)
}

\begin{frame}{The Succinctness Problem}
	To express ``some coalition can force $\varphi$'' in CL:
	\[ \bigvee_{C \subseteq \mathcal{A}g} \langle\!\langle C \rangle\!\rangle \varphi \]

	\begin{itemize}
		\item With $n$ agents, this formula grows \textbf{exponentially} ($2^n$ disjuncts).
		\item Expressing general properties like ``a majority can win'' is \textbf{practically impossible} in CL.
	\end{itemize}
\end{frame}


\begin{frame}{The Innovation: Predicates over Coalitions}
	Instead of enumerating members, describe coalitions via \textbf{predicates} $P$:

	\[
		P ::= \text{subseteq}(C) \mid \text{supseteq}(C) \mid \neg P \mid P \lor P
	\]

	\begin{description}
		\item[$\text{subseteq}(D)$] $C \models \text{subseteq}(D) \iff C \subseteq D$
		\item[$\text{supseteq}(D)$] $C \models \text{supseteq}(D) \iff C \supseteq D$
	\end{description}

	\vspace{0.5em}
	Derived predicates:
	\begin{itemize}
		\item $\text{eq}(D) \equiv \text{subseteq}(D) \land \text{supseteq}(D)$ --- exact equality
		\item $\text{incl}(i) \equiv \text{supseteq}(\{i\})$ --- agent $i$ is a member
		\item $\text{excl}(i) \equiv \neg \text{incl}(i)$ --- agent $i$ is not a member
		\item $\text{any} \equiv \text{supseteq}(\emptyset)$ --- any coalition
	\end{itemize}
\end{frame}

\note{
	Quantified Coalition Logic solves the succinctness problem by introducing a quantification mechanism
	over coalitions. Instead of enumerating the members of a coalition, we describe coalitions through
	predicates.
	The two atomic predicates are inclusion and containment: subsetof D checks whether the coalition
	is a subset of D, and supersetof D checks whether it contains D.
	From these, we can derive useful predicates: eq D for exact equality, incl i to check
	if agent i is a member, excl i for the opposite, and any which is satisfied by all coalitions.
	Note that subsetof and supersetof are mutually definable due to the finiteness of the agent set, but defining supseteq starting from subseteq
	requires an exponentially long formula, so we need both as primitivies.
}

\begin{frame}{Syntax of QCL}

	\[
		\boxed{
			\varphi ::= \top
			\mid p
			\mid \neg\varphi
			\mid \varphi\lor\varphi
			\mid \langle P\rangle\varphi
			\mid [P]\varphi
		}
	\]

	\vspace{1em}

	\begin{itemize}
		\item $\langle P\rangle\varphi$: \textbf{some} coalition satisfying $P$
		      can force $\varphi$.
		\item $[P]\varphi$: \textbf{every} coalition satisfying $P$
		      can force $\varphi$.
	\end{itemize}

\end{frame}

\begin{frame}{Semantics of QCL}

	\small

	\begin{block}{Propositional cases}
		\[
			\begin{aligned}
				\mathcal M,s \models_{\mathrm{QCL}} \top
				 &                                                      \\[0.2em]
				\mathcal M,s \models_{\mathrm{QCL}} p
				 & \iff p \in \pi(s)                                    \\[0.2em]
				\mathcal M,s \models_{\mathrm{QCL}} \neg\varphi
				 & \iff \mathcal M,s \not\models_{\mathrm{QCL}} \varphi \\[0.2em]
				\mathcal M,s \models_{\mathrm{QCL}} \varphi\lor\psi
				 & \iff
				\mathcal M,s\models_{\mathrm{QCL}}\varphi
				\text{ or }
				\mathcal M,s\models_{\mathrm{QCL}}\psi
			\end{aligned}
		\]
	\end{block}

	\begin{block}{Coalition modalities}
		\[
			\begin{aligned}
				\mathcal M,s\models_{\mathrm{QCL}}\langle P\rangle\varphi
				\iff {} &
				\exists C\subseteq\mathcal{A}g:
				C\models_{\mathrm{cp}}P
				\ \land\
				\exists S\in E(C,s): \\
				        & \qquad
				\forall s'\in S,\;
				\mathcal M,s'\models_{\mathrm{QCL}}\varphi
			\end{aligned}
		\]

		\[
			\begin{aligned}
				\mathcal M,s\models_{\mathrm{QCL}}[P]\varphi
				\iff {} &
				\forall C\subseteq\mathcal{A}g:
				C\models_{\mathrm{cp}}P
				\ \Longrightarrow\
				\exists S\in E(C,s): \\
				        & \qquad
				\forall s'\in S,\;
				\mathcal M,s'\models_{\mathrm{QCL}}\varphi
			\end{aligned}
		\]
	\end{block}

\end{frame}
